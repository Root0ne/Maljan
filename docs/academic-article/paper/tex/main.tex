% The paper. Compile with:  make paper
%
% XeLaTeX, not pdfLaTeX. The sources are UTF-8 and the em dashes, quotation
% marks and accented author names in the bibliography resolve without an
% encoding package. Mathematical symbols do not appear as literals: they are
% written as maths, and a check refuses the build if one slips through, because
% a Unicode minus is a dash-shaped glyph at text metrics and this paper reports
% several hundred signed numbers.
%
% Elsevier's elsarticle, in `preprint` — which is single column, and chosen
% rather than settled for. `5p` and `3p` set two columns for the typeset
% version; this paper carries two dozen wide tables of measurements that do not
% survive a 3.5-inch measure, and Elsevier's own guidance is that a submission
% is prepared as a preprint and typeset by the journal. Switching to `final,5p`
% later is a one-word change to this line and a rebuild of those tables, which
% is the journal's job to trigger and not a decision to pre-empt.
\documentclass[preprint,11pt,a4paper]{elsarticle}

% elsarticle loads natbib itself and numbers with elsarticle-num below.
\usepackage{fontspec}
\setmainfont{TeX Gyre Termes}
\setmonofont[Scale=0.85]{DejaVu Sans Mono}
% Maths in the body's own family. Without this, fontspec leaves maths to
% Computer Modern, so the first $-$ in the document embeds CMMI10, CMR10 and
% CMSY10 alongside Termes — three extra families, and a minus that does not
% match the digits either side of it.
\usepackage{unicode-math}
\setmathfont{TeX Gyre Termes Math}

% No geometry here, deliberately, and the reason is narrower than it first
% looked. elsarticle's own \geometry call sits inside \ifnum\jtype=1/3/5, so the
% class sets a measure for the typeset options and sets none for `preprint`. It
% leaves the article default, which is what a preprint should be. There is
% therefore nothing to override, and this file used to override it anyway: to US
% Letter at 506pt, 32% wider, so that two dozen tables laid out for the wider
% measure would fit. The tables were narrowed instead, and the page size comes
% from the class option above, because Elsevier is an A4 publisher.
\usepackage{graphicx}
% Long code literals in a narrow table cell have no break point of their own:
% TeX will not break at an underscore or a slash, and a 38-character model
% identifier ran 132pt past its column. seqsplit gives the cell somewhere to
% break; it is applied only where a literal is too long for its column.
\usepackage{seqsplit}
\usepackage{booktabs}
\usepackage{array}
\usepackage{calc}
% Expansion is a pdftex feature; XeTeX refuses it outright. Protrusion works
% and is what microtype can contribute to justification here.
\usepackage[protrusion=true]{microtype}
\usepackage{xcolor}
\usepackage{amsmath}
\usepackage{fancyvrb}
\usepackage[hidelinks]{hyperref}
% A section title containing maths becomes a PDF bookmark, and unicode-math's
% symbols are not expandable in that string — the build died on "Improper
% alphabetic constant" the first time a heading carried a multiplication sign.
\pdfstringdefDisableCommands{%
  \def\times{x}\def\Delta{Delta}\def\rho{rho}\def\alpha{alpha}%
  \def\beta{beta}\def\sigma{sigma}\def\mu{mu}%
  \def\ge{>=}\def\le{<=}\def\gg{>>}\def\ll{<<}%
  \def\approx{~}\def\neq{!=}\def\pm{+/-}%
}

% Long code literals in narrow table columns overrun their neighbour. Verbatim
% text has no discretionary break points, so a cell wider than its column simply
% keeps going; allowing breaks inside \texttt fixes the class rather than the
% instances.
\usepackage[htt]{hyphenat}
% Verbatim blocks are transcripts of real output and cannot be re-wrapped
% without falsifying them, so they are set smaller rather than broken. The
% widest is 79 characters and the measure no longer carries that at body size.
\makeatletter
\def\verbatim@font{\ttfamily\scriptsize}
\makeatother
% Figures are drawn wider than the measure. Bound them to it rather than
% trusting each figure's own size.
\setkeys{Gin}{width=\linewidth,keepaspectratio}
% Float placement. LaTeX's defaults reserve most of a page for text and cap how
% many floats a page may carry, and with two dozen tables that meant every one
% of them was deferred to the end of the document: the first table appeared on
% page 69 of 80, sixty pages from the sentence it belongs to. These are the
% standard relaxations, and they put each table back on or beside its own page.
\setcounter{topnumber}{3}
\setcounter{bottomnumber}{2}
\setcounter{totalnumber}{5}
\renewcommand{\topfraction}{0.9}
\renewcommand{\bottomfraction}{0.7}
\renewcommand{\textfraction}{0.07}
\renewcommand{\floatpagefraction}{0.75}
% Rather than \sloppy, which sets tolerance to 9999 and emergencystretch to 3em
% and buys a clean right margin with visibly loose lines. At the class measure
% that trade produced eleven underfull lines; these numbers are the tightest
% setting that still reaches zero overfull.
\tolerance=1800
\emergencystretch=1.5em
% \sloppy sets \hfuzz=0.5pt, which stops sub-half-point overruns being reported
% at all. A zero-overfull log bought that way is not a zero-overfull document,
% so the tolerance goes back to nothing and the zero has to be earned.
\hfuzz=0pt
\vfuzz=0pt

% No \captionsetup and no \parskip either, for the same reason. elsarticle
% defines its own \@makecaption, at footnotesize, "Table 1: text", label not
% bold, and setting labelfont=bf overrode the class to put bold on the page in
% the one place a reader cannot avoid it.
\graphicspath{{../}{./}{../figures/}}
